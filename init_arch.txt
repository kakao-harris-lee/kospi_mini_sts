def publish_raw_data(data):
    """
    API 콜백 함수에 연결되어 실시간으로 데이터를 Stream에 추가
    """
    try:
        # data는 API로부터 받은 틱/호가 정보 딕셔너리
        r.xadd(
            RAW_STREAM, 
            {'timestamp': time.time(), 'data_json': json.dumps(data)}, 
            maxlen=10000,  # 메모리 관리를 위해 최대 길이 설정 (FIFO)
            limit=False
        )
    except Exception as e:
        print(f"Error publishing to Redis: {e}")
